{{/* Partial: media/_ Purpose: Render an image, local video, or embedded video from normalized media data. Context: - img (string, optional): Image path or URL. - vid (string, optional): Video path or supported provider URL. - alt (string, optional): Alternative text; derived from the media source when empty. Renders: The matching image, video, or embedded-video markup. Side effects: May process local image resources; config.imgs_remote opts into cached external-image downloads. */}} {{- $src := or .vid .img -}} {{- if $src -}} {{- $media := . -}} {{- $img := .img -}} {{- $vid := .vid -}} {{- $alt := .alt -}} {{- if and .img (not .vid) -}} {{- if partial "func/is-vid" .img -}} {{- $vid = .img -}} {{- $img = "" -}} {{- end -}} {{- end -}} {{- if and $vid (not $img) -}} {{- $img = partial "func/poster" $vid -}} {{- end -}} {{- if $vid -}} {{- $opts := merge . (dict "vid" $vid "poster" $img) -}} {{- $vid = partial "media/vid-data" $opts -}} {{- else -}} {{- $vid = dict -}} {{- end -}} {{- with $img -}} {{- $img = partial "func/check-media" . -}} {{- end -}} {{- if .alt -}} {{- $alt = string .alt -}} {{- else if or .vid_title .video_title -}} {{- $alt = or .vid_title .video_title -}} {{- else if not .alt -}} {{- if and (index $vid "ifr") (index $vid "meta" "name") -}} {{- $alt = $vid.meta.name -}} {{- else if index $vid "ifr" -}} {{- $alt = print (i18n "vid") " · " $vid.name " · " $vid.id -}} {{- else -}} {{- $src = or (index $vid "url") $img -}} {{- $alt = path.Base ((urls.Parse $src).Path) | replaceRE `\.[^.]+$` "" | humanize -}} {{- end -}} {{- end -}} {{- $lazy := cond (not .notlazy) "lazy" "" -}} {{- $img_ext := hasPrefix $img "http" -}} {{- $remote := dict -}} {{- if and $img_ext hugo.Data.config.imgs_remote -}} {{- $remote = partialCached "media/remote-resource" $img $img -}} {{- with $remote.resource -}} {{- $img = .RelPermalink -}} {{- $img_ext = false -}} {{- end -}} {{- end -}} {{- $intrinsic := index $vid "presentation" | default dict -}} {{- if not $vid -}} {{- $resource := false -}} {{- if $remote.resource -}} {{- $resource = $remote.resource -}} {{- else if not $img_ext -}} {{- $resource = resources.Get $img -}} {{- end -}} {{- if $remote.resource -}} {{- $intrinsic = $remote.dimensions -}} {{- else -}} {{- with $resource -}} {{- if eq .MediaType.SubType "svg" -}} {{- $size := partial "func/svg-sizes" .Content -}} {{- $intrinsic = dict "width" $size.width "height" $size.height -}} {{- else if eq .ResourceType "image" -}} {{- $intrinsic = dict "width" .Width "height" .Height -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $dimensions := partial "media/dimensions" (merge . (dict "intrinsic" $intrinsic)) -}} {{- if $vid -}}{{- $vid = merge $vid (dict "presentation" $dimensions) -}}{{- end -}} {{- $media_new := dict "img" $img "vid" $vid "alt" $alt "lazy" $lazy "img_ext" $img_ext "dimensions" $dimensions "resource" $remote.resource "remote_svg" $remote.svg -}} {{- $media = merge $media $media_new -}} {{- if or (index $vid "mute") (and $vid (not $img)) -}} {{- partial "media/vid" $media -}} {{- else if or $img_ext $remote.svg -}} {{- partial "media/out-img" $media -}} {{- else if hasSuffix $img ".svg" -}} {{- partial "media/in-svg" $media -}} {{- else -}} {{- partial "media/in-img" $media -}} {{- end -}} {{- end -}}
