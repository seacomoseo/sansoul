{{/* Partial: media/_ Purpose: Render an image, local video, or embedded video from normalized media data. Context: - img (string, optional): Image path or URL. - vid (string, optional): Video path or supported provider URL. - alt (string, optional): Alternative text; derived from the media source when empty. Renders: The matching image, video, or embedded-video markup. Side effects: May process local image resources. */}} {{- $src := or .vid .img -}} {{- if $src -}} {{- $media := . -}} {{- $img := .img -}} {{- $vid := .vid -}} {{- $alt := .alt -}} {{- if and .img (not .vid) -}} {{- if partial "func/is-vid" .img -}} {{- $vid = .img -}} {{- $img = "" -}} {{- end -}} {{- end -}} {{- if and $vid (not $img) -}} {{- $img = partial "func/poster" $vid -}} {{- end -}} {{- if $vid -}} {{- $opts := merge . (dict "vid" $vid "poster" $img) -}} {{- $vid = partial "media/vid-data" $opts -}} {{- else -}} {{- $vid = dict -}} {{- end -}} {{- with $img -}} {{- $img = partial "func/check-media" . -}} {{- end -}} {{- if .alt -}} {{- $alt = string .alt -}} {{- else if or .vid_title .video_title -}} {{- $alt = or .vid_title .video_title -}} {{- else if not .alt -}} {{- if and (index $vid "ifr") (index $vid "meta" "name") -}} {{- $alt = $vid.meta.name -}} {{- else if index $vid "ifr" -}} {{- $alt = print (i18n "vid") " · " $vid.name " · " $vid.id -}} {{- else -}} {{- $src = or (index $vid "url") $img -}} {{- $alt = path.Base ((urls.Parse $src).Path) | replaceRE `\.[^.]+$` "" | humanize -}} {{- end -}} {{- end -}} {{- $lazy := cond (not .notlazy) "lazy" "" -}} {{- $img_ext := hasPrefix $img "http" -}} {{- $media_new := dict "img" $img "vid" $vid "alt" $alt "lazy" $lazy "img_ext" $img_ext -}} {{- $media = merge $media $media_new -}} {{- if or (index $vid "mute") (and $vid (not $img)) -}} {{- partial "media/vid" $media -}} {{- else if $img_ext -}} {{- partial "media/out-img" $media -}} {{- else if hasSuffix $img ".svg" -}} {{- partial "media/in-svg" $media -}} {{- else -}} {{- partial "media/in-img" $media -}} {{- end -}} {{- end -}}
