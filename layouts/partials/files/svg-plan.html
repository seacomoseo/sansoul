{{/* Build and publish SVG resources once, after every page and output format has rendered. Group membership is derived from the output URLs that use each file. */}} {{- $format := .format | default "html" -}} {{- $use_key := printf "sansoul.resources.%s.svg-uses" $format -}} {{- $image_key := printf "sansoul.resources.%s.svg-images" $format -}} {{- $svg_key := printf "sansoul.resources.%s.svgs" $format -}} {{- $div_key := printf "sansoul.resources.%s.divs" $format -}} {{- $glyph_key := printf "sansoul.resources.%s.glyphs" $format -}} {{- $use_records := slice -}} {{- $image_records := slice -}} {{- $div_records := slice -}} {{- $glyphs := slice -}} {{- $rendered_usage := hugo.Store.Get (print "sansoul.resources.rendered-usage." $format) | default dict -}} {{- range hugo.Sites -}} {{- range .Pages -}} {{- $source_page := . -}} {{- if eq ($source_page.Store.Get "sansoul.output-format" | default "html") $format -}} {{- $output := $source_page.RelPermalink -}} {{- $usage := index $rendered_usage $output | default dict -}} {{- range $usage.uses | default (slice) -}} {{- $use_records = $use_records | append (dict "name" . "page" $output) -}} {{- end -}} {{- range $usage.images | default (slice) -}} {{- $image_records = $image_records | append (dict "name" . "page" $output) -}} {{- end -}} {{- range $usage.divs | default (slice) -}} {{- $div_records = $div_records | append (dict "name" . "page" $output) -}} {{- end -}} {{- range $source_page.Store.Get $glyph_key | default (slice) -}} {{- $glyphs = $glyphs | append . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $use_names := slice -}} {{- range $use_records -}}{{- $use_names = $use_names | append .name -}}{{- end -}} {{- $use_names = $use_names | uniq | sort -}} {{- $image_names := slice -}} {{- range $image_records -}}{{- $image_names = $image_names | append .name -}}{{- end -}} {{- $image_names = $image_names | uniq | sort -}} {{- $all_names := append $use_names $image_names | uniq | sort -}} {{- $div_names := slice -}} {{- range $div_records -}}{{- $div_names = $div_names | append .name -}}{{- end -}} {{- $div_names = $div_names | uniq | sort | complement (slice "fade") -}} {{- $glyphs = $glyphs | uniq | sort -}} {{- $items := slice -}} {{- $id_records := slice -}} {{- range $all_names -}} {{- $name := . -}} {{- $id := partial "func/svg-fragment-id" $name -}} {{- $resource_id := partial "func/svg-resource-id" $name -}} {{- $is_image := in $image_names $name -}} {{- $view_id := print "sansoul-image-view-" $id -}} {{- $path := print "/u/{icons/,}" $name ".svg" -}} {{- $source := resources.GetMatch $path -}} {{- if not $source -}} {{- errorf "SanSoul SVG resource %q was registered but no matching file exists at %s" $name $path -}} {{- else -}} {{- $content := ($source | minify).Content | replaceRE `^<\?xml[^>]*\?>` "" -}} {{- $root := findRE `^]*>` $content -}} {{- $root_id := "" -}} {{- with $root -}} {{- with findRESubmatch `(?:^|\s)id=["']([^"']+)["']` (index . 0) -}} {{- $root_id = index (index . 0) 1 -}} {{- end -}} {{- end -}} {{- $ids := slice -}} {{- range findRESubmatch `\bid=["']([^"']+)["']` $content -}} {{- $ids = $ids | append (index . 1) -}} {{- end -}} {{- if not (in $ids $id) -}} {{- $ids = $ids | append $id -}} {{- else if ne $root_id $id -}} {{- $ids = $ids | append $id -}} {{- end -}} {{- if $is_image -}}{{- $ids = $ids | append $view_id -}}{{- end -}} {{- $unsafe_root_ref := and $root_id (ne $root_id $id) (strings.Contains $content (print "#" $root_id)) -}} {{- $unsafe_style := findRE `(?i))` $content -}} {{- $animated := findRE `(?i)<(animate(?:Motion|Transform)?|set)\b|@keyframes|\banimation(-name)?\s*:` $content -}} {{- $items = $items | append (dict "name" $name "id" $id "resource_id" $resource_id "view_id" $view_id "source" $source "ids" $ids "unsafe" (or $unsafe_root_ref $unsafe_style) "animated" $animated "image" $is_image) -}} {{- $id_records = $id_records | append (dict "name" $name "ids" $ids "unsafe" (or $unsafe_root_ref $unsafe_style)) -}} {{- end -}} {{- end -}} {{- /* A combined SVG document has one ID namespace; isolate every asset in a collision. */ -}} {{- $owners := dict -}} {{- $colliding_names := slice -}} {{- range $id_records -}} {{- $name := .name -}} {{- if .unsafe -}}{{- $colliding_names = $colliding_names | append $name -}}{{- end -}} {{- $seen := slice -}} {{- range .ids -}} {{- if in $seen . -}} {{- $colliding_names = $colliding_names | append $name -}} {{- else -}} {{- $seen = $seen | append . -}} {{- end -}} {{- if isset $owners . -}} {{- $colliding_names = $colliding_names | append $name (index $owners .) -}} {{- else -}} {{- $owners = merge $owners (dict . $name) -}} {{- end -}} {{- end -}} {{- end -}} {{- $colliding_names = $colliding_names | uniq -}} {{- $references := dict -}} {{- $div_references := dict -}} {{- $individual := dict -}} {{- range $items -}} {{- if in $colliding_names .name -}} {{- $individual = merge $individual (dict .name .) -}} {{- end -}} {{- end -}} {{- range $name, $item := $individual -}} {{- $resource := partial "files/svg-resource" (dict "source" $item.source "name" $name "id" $item.id) -}} {{- $token := print "__SANSOUL_SVG_" ($item.resource_id | urlize) "__" -}} {{- $image_token := print "__SANSOUL_SVG_IMAGE_" ($item.resource_id | urlize) "__" -}} {{- $references = merge $references (dict $token $resource.url $image_token $resource.url) -}} {{- end -}} {{- /* Ordinary-use sprites only combine assets with the same rendered-page set. */ -}} {{- $use_groups := dict -}} {{- $use_signatures := slice -}} {{- range $use_names -}} {{- $name := . -}} {{- if not (in $colliding_names $name) -}} {{- $asset_records := where $use_records "name" $name -}} {{- $asset_pages := slice -}} {{- range $asset_records -}}{{- $asset_pages = $asset_pages | append .page -}}{{- end -}} {{- $asset_pages = $asset_pages | uniq | sort -}} {{- $signature := jsonify $asset_pages | string -}} {{- $item := index (where $items "name" $name) 0 -}} {{- if and (gt (len $asset_pages) 1) (not $item.animated) -}} {{- $group_items := index $use_groups $signature | default (slice) -}} {{- $use_groups = merge $use_groups (dict $signature ($group_items | append $item)) -}} {{- $use_signatures = $use_signatures | append $signature -}} {{- end -}} {{- end -}} {{- end -}} {{- $use_signatures = $use_signatures | uniq | sort -}} {{- range $signature := $use_signatures -}} {{- $group_items := index $use_groups $signature -}} {{- $group_id := substr (sha256 $signature) 0 12 -}} {{- $sprite := partial "files/svg-sprite" (dict "group" (print "usage-" $group_id) "items" $group_items) -}} {{- range $group_items -}} {{- $token := print "__SANSOUL_SVG_" (.resource_id | urlize) "__" -}} {{- $references = merge $references (dict $token (print $sprite.url "#" .id)) -}} {{- end -}} {{- end -}} {{- /* Image sprites also follow exact page-use signatures and retain semantics. */ -}} {{- $image_groups := dict -}} {{- $image_signatures := slice -}} {{- range $image_names -}} {{- $name := . -}} {{- if not (in $colliding_names $name) -}} {{- $asset_records := where $image_records "name" $name -}} {{- $asset_pages := slice -}} {{- range $asset_records -}}{{- $asset_pages = $asset_pages | append .page -}}{{- end -}} {{- $signature := jsonify ($asset_pages | uniq | sort) | string -}} {{- $group_items := index $image_groups $signature | default (slice) -}} {{- $item := index (where $items "name" $name) 0 -}} {{- $image_groups = merge $image_groups (dict $signature ($group_items | append $item)) -}} {{- $image_signatures = $image_signatures | append $signature -}} {{- end -}} {{- end -}} {{- $image_signatures = $image_signatures | uniq | sort -}} {{- range $signature := $image_signatures -}} {{- $group_items := index $image_groups $signature -}} {{- $image_sprite := partial "files/svg-image-sprite" (dict "group" $signature "items" $group_items) -}} {{- range $group_items -}} {{- $token := print "__SANSOUL_SVG_IMAGE_" (.resource_id | urlize) "__" -}} {{- $image_url := print $image_sprite.RelPermalink "#sansoul-image-view-" .id -}} {{- $references = merge $references (dict $token $image_url) -}} {{- $use_token := print "__SANSOUL_SVG_" (.resource_id | urlize) "__" -}} {{- if not (index $references $use_token) -}} {{- $references = merge $references (dict $use_token $image_url) -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Divider sheet groups are based on the same output-page signatures. */ -}} {{- $div_groups := dict -}} {{- $div_signatures := slice -}} {{- range $div_names -}} {{- $name := . -}} {{- $asset_records := where $div_records "name" $name -}} {{- $asset_pages := slice -}} {{- range $asset_records -}}{{- $asset_pages = $asset_pages | append .page -}}{{- end -}} {{- $signature := jsonify ($asset_pages | uniq | sort) | string -}} {{- $group_divs := index $div_groups $signature | default (slice) -}} {{- $div_groups = merge $div_groups (dict $signature ($group_divs | append $name)) -}} {{- $div_signatures = $div_signatures | append $signature -}} {{- end -}} {{- $div_signatures = $div_signatures | uniq | sort -}} {{- range $signature := $div_signatures -}} {{- $group_divs := index $div_groups $signature | uniq | sort -}} {{- $content := partial "files/draws-svg" (dict "divs" $group_divs) | string -}} {{- $target_id := substr (sha256 $signature) 0 12 -}} {{- $source_target := print "draws/group-" $target_id ".svg" -}} {{- $resource := $content | resources.FromString $source_target -}} {{- if hugo.IsProduction -}}{{- $resource = $resource | minify -}}{{- end -}} {{- $resource = partial "func/publish-content-resource" (dict "resource" $resource "target" "draws/group.svg") -}} {{- range $group_divs -}} {{- $reference_key := print $signature "|" . -}} {{- $div_references = merge $div_references (dict $reference_key $resource.RelPermalink) -}} {{- end -}} {{- end -}} {{- /* Material Symbols uses the final union, after all page Stores are complete. */ -}} {{- $glyphs_html := partial "files/glyphs" (dict "format" $format "preload" true "glyphs" $glyphs) -}} {{- return (dict "format" $format "items" $items "references" $references "div-references" $div_references "div-records" $div_records "colliding" $colliding_names "use-records" $use_records "image-records" $image_records "glyphs-html" $glyphs_html) -}}
