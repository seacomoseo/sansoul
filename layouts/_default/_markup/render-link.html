{{- /* Link metadata is a recognized token block at the end of the title. Legacy square blocks and nofollow remain readable; CMS writes braces and lock. */ -}} {{- $label := .Text -}} {{- $anchor := .Title -}} {{- $title := "" -}} {{- $meta := slice -}} {{- $token := `(main|alt|cta|light|dark|turn|such|whatsapp|dot|sm|swap|lock|nofollow|sublock|blank|ga4(:[A-Za-z][A-Za-z0-9_]{0,39}(,[A-Za-z][A-Za-z0-9_]{0,39})*)?)` -}} {{- $pattern := printf `\s*(\{\s*%s(\s+%s)*\s*\}|\[\s*%s(\s+%s)*\s*\])$` $token $token $token $token -}} {{- with findRE $pattern .Title 1 -}} {{- $block := index . 0 -}} {{- $anchor = strings.TrimSpace (strings.TrimSuffix $block $.Title) -}} {{- $meta = findRE `\S+` (replaceRE `[\[\]{}]` "" $block) -}} {{- end -}} {{- /* button */ -}} {{- $btn := "" -}} {{- $dot := "" -}} {{- $sm := "" -}} {{- $swap := "" -}} {{- if and (hasPrefix .Text "[") (hasSuffix .Text "]") -}} {{- $label = .Text | strings.TrimPrefix "[" | strings.TrimSuffix "]" -}} {{- $btn = "cta" -}} {{- range $color := slice "main" "alt" "cta" "light" "dark" "turn" "such" "whatsapp" -}} {{- if in $meta $color -}}{{- $btn = $color -}}{{- end -}} {{- end -}} {{- $dot = in $meta "dot" -}} {{- $sm = in $meta "sm" -}} {{- $swap = in $meta "swap" -}} {{- end -}} {{- /* ga4 */ -}} {{- $ga4 := "" -}} {{- range $meta -}} {{- if eq . "ga4" -}}{{- $ga4 = "contact" -}} {{- else if hasPrefix . "ga4:" -}}{{- $ga4 = strings.TrimPrefix "ga4:" . -}}{{- end -}} {{- end -}} {{- /* blank */ -}} {{- $blank := in $meta "blank" -}} {{- /* icon outside */ -}} {{- $outside := partial "func/link-icons" $label -}} {{- $label := $outside.text -}} {{- $il := $outside.il -}} {{- $ir := $outside.ir -}} {{- /* lock */ -}} {{- $lock := cond (or (in $meta "lock") (in $meta "nofollow")) true nil -}} {{- if in $meta "sublock" -}} {{- if not page.IsHome -}} {{- $lock = true -}} {{- end -}} {{- end -}} {{- if $lock -}} {{- /* change anchor by title and remove */ -}} {{- $title = $anchor -}} {{- $anchor = "" -}} {{- end -}} {{- /* link */ -}} {{- partial "link" (dict "btn" $btn "dot" $dot "sm" $sm "swap" $swap "ga4" $ga4 "il" $il "ir" $ir "label" $label "anchor" $anchor "title" $title "url" .Destination "lock" $lock "blank" $blank ) -}}
